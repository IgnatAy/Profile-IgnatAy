'use client';

import { useCallback, useEffect, useRef, useState } from 'react';
import { ArrowUpRight, Check, Eye, Layers, MessageCircle, Moon, Pause, Play, Send, Sparkles } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Switch } from '@/components/ui/switch';
import { Slider } from '@/components/ui/slider';
import { AliceCharacter, type AlicePose } from '@/models/alice/alice-character';
import { registerCharacterTool } from '@/lib/character-tools';

const poses: { id: AlicePose; label: string; sub: string; line: string }[] = [
  { id: 'idle', label: '安静陪伴', sub: 'IDLE', line: '你来了。随便坐吧。' },
  { id: 'thinking', label: '想一想', sub: 'THINKING', line: '……让我想一想。' },
  { id: 'front', label: '看向你', sub: 'HELLO', line: '嗯？有什么想说的吗？' },
  { id: 'shy', label: '靠近一点', sub: 'SHY', line: '……靠得太近了。' },
];
export type CharacterController = {
  setState: (state: AlicePose) => void;
  say: (text: string) => void;
  setMotion: (enabled: boolean) => void;
  blink: () => void;
  setLayered: (enabled: boolean) => void;
  setExploded: (enabled: boolean) => void;
};
declare global { interface Window { character?: CharacterController } }

export default function Home() {
  const [pose, setPose] = useState<AlicePose>('idle');
  const [motion, setMotion] = useState(true);
  const [line, setLine] = useState('你来了。随便坐吧。');
  const [speaking, setSpeaking] = useState(false);
  const [input, setInput] = useState('');
  const [ready, setReady] = useState(false);
  const [error, setError] = useState(false);
  const [fullBody, setFullBody] = useState(false);
  const [layered, setLayered] = useState(true);
  const [exploded, setExploded] = useState(false);
  const [strength, setStrength] = useState(1.4);
  const [blinkSignal, setBlinkSignal] = useState(0);
  const speechTimer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const tapCount = useRef(0);
  const say = useCallback((text: string) => {
    const value = String(text).trim().slice(0, 100);
    if (!value) return;
    if (speechTimer.current) clearTimeout(speechTimer.current);
    setLine(value);
    setSpeaking(true);
    speechTimer.current = setTimeout(() => setSpeaking(false), Math.min(7000, Math.max(1600, value.length * 150)));
  }, []);
  const selectPose = useCallback((state: AlicePose) => {
    const next = poses.find(p => p.id === state);
    if (!next) return;
    if (speechTimer.current) clearTimeout(speechTimer.current);
    setSpeaking(false);
    setPose(state);
    setExploded(false);
    setLine(next.line);
  }, []);
  const blink = useCallback(() => setBlinkSignal(value => value + 1), []);
  const changeLayered = useCallback((enabled: boolean) => {
    setLayered(enabled); setExploded(false);
  }, []);
  const changeExploded = useCallback((enabled: boolean) => {
    if (enabled) setLayered(true);
    setExploded(enabled);
  }, []);
  useEffect(() => {
    const api = { setState: selectPose, say, setMotion, blink, setLayered: changeLayered, setExploded: changeExploded };
    window.character = api;
    const cleanupTools = registerCharacterTool(api);
    return () => {
      cleanupTools();
      if (window.character === api) delete window.character;
      if (speechTimer.current) clearTimeout(speechTimer.current);
    };
  }, [say, selectPose, blink, changeLayered, changeExploded]);
  useEffect(() => {
    const query = matchMedia('(prefers-reduced-motion: reduce)');
    const update = () => { if (query.matches) { setMotion(false); } };
    update(); query.addEventListener('change', update);
    return () => query.removeEventListener('change', update);
  }, []);
  const greet = () => {
    say(['下午好。今天也在忙自己的事情吗？', '这里很安静。待一会儿也可以。', '我在听。', '……茶要凉了。'][tapCount.current++ % 4]);
  };

  return (
    <main className="winter-room">
      <header className="room-header">
        <div className="room-brand"><span className="brand-symbol">A</span><span>有珠的冬日小屋</span></div>
        <span className="edition">INTERACTIVE PORTRAIT <span>02</span></span>
      </header>
      <div className="room-layout">
        <section className="control-panel" aria-label="角色交互">
          <div className="intro">
            <p className="eyebrow"><span /> A QUIET MOMENT</p>
            <h1>有珠<span className="name-divider"> / </span><span className="latin-name">Alice</span></h1>
            <p className="intro-note">在你的网页里，安静地陪伴。</p>
          </div>
          <div className="dialogue" aria-live="polite" aria-atomic="true">
            <span className="dialogue-name">久远寺有珠 <span>ALICE KUONJI</span></span>
            <p key={line}>{line}</p>
            <span className="dialogue-indicator" aria-hidden="true">{speaking ? '•••' : '◇'}</span>
          </div>
          <div className="pose-label"><span>此刻的她</span><span className="section-number">01 — 04</span></div>
          <div className="pose-grid">
            {poses.map((p, i) => <Button key={p.id} variant="ghost" className={`pose-button ${pose === p.id ? 'is-selected' : ''}`} aria-pressed={pose === p.id} disabled={!ready || error} onClick={() => selectPose(p.id)}>
              <span className="pose-index">0{i + 1}</span><span className="pose-copy"><span>{p.label}</span><small>{p.sub}</small></span>
              {pose === p.id ? <Check size={16} /> : <ArrowUpRight size={16} />}
            </Button>)}
          </div>
          <div className="motion-controls">
            <label htmlFor="layered-toggle"><span><Layers size={16} />使用拆层模型</span><Switch id="layered-toggle" checked={layered} onCheckedChange={changeLayered} /></label>
            <label htmlFor="motion-toggle"><span><Sparkles size={16} />{layered ? pose === 'thinking' ? '呼吸与轻微摆动' : '呼吸、眨眼与轻微摆动' : '轻微呼吸与眼神'}</span><Switch id="motion-toggle" checked={motion} onCheckedChange={setMotion} /></label>
            {layered && <div className="rig-settings">
              <div className="rig-actions"><Button variant="outline" onClick={blink} disabled={pose === 'thinking'}><Eye size={16} />{pose === 'thinking' ? '闭眼沉思' : '眨一下眼'}</Button><Button variant="outline" aria-pressed={exploded} onClick={() => changeExploded(!exploded)}><Layers size={16} />{exploded ? '合回模型' : '查看拆层'}</Button></div>
              <label id="strength-label"><span>动作幅度</span><span>{strength.toFixed(1)}×</span></label>
              <Slider aria-labelledby="strength-label" value={[Math.round(strength * 100)]} min={30} max={180} step={10} onValueChange={value => setStrength((Array.isArray(value) ? value[0] : value) / 100)} />
            </div>}
            <p className="rig-note">{layered ? exploded ? '头部、发束、蝴蝶结、五官与身体分开展示。' : pose === 'thinking' ? '保留闭眼思考，手与下巴一起随动。' : '当前姿态已拆层；关闭开关可对比原画。' : '整图版本 · 开启拆层模型可对比独立运动。'}</p>
          </div>
          <form className="say-form" onSubmit={e => { e.preventDefault(); if (!input.trim()) return; say(input); setInput(''); }}>
            <label htmlFor="say-input">让她说一句话<span>文字演示 · 无语音</span></label>
            <div className="say-field"><input id="say-input" value={input} onChange={e => setInput(e.target.value)} maxLength={100} placeholder="例如：欢迎来到我的主页。" autoComplete="off" /><Button type="submit" size="icon" disabled={!input.trim() || !ready} aria-label="展示这句话"><Send size={17} /></Button></div>
          </form>
        </section>
        <section className={`portrait-stage ${fullBody ? 'is-full-body' : ''}`} aria-label="有珠动态立绘">
          <div className="stage-top"><span className="live-status"><i />{error ? '素材加载失败' : ready ? '她在这里' : '正在准备立绘'}</span><Button variant="ghost" onClick={() => setFullBody(!fullBody)} className="view-toggle">{fullBody ? '切换半身' : '看看全身'}<ArrowUpRight size={14} /></Button></div>
          <div className="portrait-orbit" aria-hidden="true" />
          <div className="portrait-word" aria-hidden="true">ALICE</div>
          <AliceCharacter pose={pose} speaking={speaking} motion={motion} autonomous={motion} fullBody={fullBody} layered={layered} exploded={exploded} strength={strength} blinkSignal={blinkSignal} onInteract={greet} onReady={() => setReady(true)} onError={() => setError(true)} onLayerError={() => { setLayered(false); setLine('拆层素材暂时无法加载，已切回原画版本。'); }} />
          {error && <div className="asset-error">立绘暂时没有载入。<Button onClick={() => location.reload()}>重新加载</Button></div>}
          <div className="stage-bottom"><span><MessageCircle size={15} />点一下她，打个招呼</span><Button variant="ghost" size="icon" aria-label={motion ? '暂停待机动画' : '播放待机动画'} onClick={() => setMotion(!motion)}>{motion ? <Pause size={15} /> : <Play size={15} />}</Button></div>
        </section>
      </div>
      <footer className="room-footer"><span><Moon size={14} />四种姿态拆层 · 独立运动预览</span><span>WINTER, IN GOOD COMPANY.</span></footer>
    </main>
  );
}
